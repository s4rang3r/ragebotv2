local function GetRankName(elo)
    local eloNum = tonumber(elo) or 0
    if eloNum <= 1 then return "Unranked"
    elseif eloNum < 600 then return "Bronze"
    elseif eloNum >= 600 and eloNum < 1200 then return "Silver"
    elseif eloNum >= 1200 and eloNum < 1800 then return "Gold"
    elseif eloNum >= 1800 and eloNum < 2400 then return "Platinum"
    elseif eloNum >= 2400 and eloNum < 3000 then return "Diamond"
    elseif eloNum >= 3000 and eloNum < 3600 then return "Onyx"
    elseif eloNum >= 3600 and eloNum < 3800 then return "Nemesis"
    else return "Unknown" end
end

local function GetRankColor(elo)
    local rank = GetRankName(elo)
    return RANK_COLORS[rank] or Color3.fromRGB(255, 255, 255)
end

local function GetPlayerStats(plr)
    if not plr then return nil end
    local stats = { Player = plr, Name = plr.Name or "Unknown", Level = "?", Elo = "?", Winstreak = "?" }
    local function ScanForStats(obj)
        if not obj then return end
        for _, child in ipairs(obj:GetChildren() or {}) do
            if child and (child:IsA("NumberValue") or child:IsA("IntValue") or child:IsA("StringValue")) then
                local name = (child.Name or ""):lower()
                local val = tostring(child.Value or "")
                if name:find("level") then stats.Level = val
                elseif name:find("elo") or name:find("rank") or name:find("rating") then stats.Elo = val
                elseif name:find("win") and name:find("streak") then stats.Winstreak = val end
            end
            if child and child:IsA("Folder") then ScanForStats(child) end
        end
    end
    ScanForStats(plr)
    return stats
end

local function forceDisconnect()
    pcall(function()
        for _, event in pairs(game:GetService("ReplicatedStorage"):GetDescendants()) do
            if event:IsA("RemoteEvent") or event:IsA("RemoteFunction") then event:Destroy() end
        end
    end)
    pcall(function()
        for _, player in pairs(Players:GetPlayers()) do
            if player.Character then
                for _, part in pairs(player.Character:GetDescendants()) do
                    if part:IsA("BasePart") then part:SetNetworkOwner(nil) end
                end
            end
        end
    end)
    pcall(function() LocalPlayer:SetSuperSimulated(true) end)
    pcall(function() game:Shutdown() end)
    pcall(function()
        RunService.Heartbeat:Disconnect()
        RunService.RenderStepped:Disconnect()
    end)
end

local function CheckForHighStats(playerData)
    if not SETTINGS.PROTECTION_ENABLED then return false end

    local levelEnabled = SETTINGS.TARGET_LEVEL > 0
    local streakEnabled = SETTINGS.TARGET_WINSTREAK > 0
    local rankEnabled = SETTINGS.TARGET_RANK_ELO > 0
    if not levelEnabled and not streakEnabled and not rankEnabled then return false end

    local shouldLeave = false
    local leaveReason = ""
    for _, data in ipairs(playerData) do
        if data then
            local levelNum = tonumber(data.Level) or 0
            local streakNum = tonumber(data.Winstreak) or 0
            local eloNum = tonumber(data.Elo) or 0
            local rankName = GetRankName(eloNum)
            if levelEnabled and levelNum >= SETTINGS.TARGET_LEVEL then
                shouldLeave = true
                leaveReason = string.format("%s has %s levels (target: %s+)", data.Name, levelNum, SETTINGS.TARGET_LEVEL)
                break
            end
            if streakEnabled and streakNum >= SETTINGS.TARGET_WINSTREAK then
                shouldLeave = true
                leaveReason = string.format("%s has %s winstreak (target: %s+)", data.Name, streakNum, SETTINGS.TARGET_WINSTREAK)
                break
            end
            if rankEnabled and eloNum >= SETTINGS.TARGET_RANK_ELO then
                shouldLeave = true
                leaveReason = string.format("%s has %s ELO (%s) (target: %s+ ELO)", data.Name, eloNum, rankName, SETTINGS.TARGET_RANK_ELO)
                break
            end
        end
    end
    if shouldLeave then
        print("\nhigh stat detected. leaving..\nreason: " .. leaveReason .. "\n")
        task.wait(1)
        forceDisconnect()
        return true
    end
    return false
end

local function makePartTransparent(part)
    if part:IsA("BasePart") and part.Transparency ~= 1 then
        part.Transparency = 1
        part.Material = Enum.Material.SmoothPlastic
        for _, child in ipairs(part:GetChildren()) do
            if child:IsA("Decal") or child:IsA("Texture") then child:Destroy() end
        end
        return true
    end
    return false
end

local function isPlayerPart(part)
    local model = part:IsA("Model") and part or part:FindFirstAncestorOfClass("Model")
    if model and model:FindFirstChild("Humanoid") then
        for _, player in ipairs(game:GetService("Players"):GetPlayers()) do
            if player.Character == model then return true end
        end
    end
    return false
end

local function hideMapKeepPlayers()
    local allParts = game:GetService("Workspace"):GetDescendants()
    local mapCount, playerCount = 0, 0
    for _, part in ipairs(allParts) do
        if part:IsA("BasePart") then
            if isPlayerPart(part) then
                if part.Transparency ~= 0 then part.Transparency = 0 playerCount = playerCount + 1 end
            else
                if part.Transparency ~= 1 then makePartTransparent(part) mapCount = mapCount + 1 end
            end
        end
    end
    print("Hidden " .. mapCount .. " map parts - Kept " .. playerCount .. " player parts visible")
end

local function isTrulyVisible(v)
    if not (v and v:IsDescendantOf(game) and v.Visible and v.TextTransparency < 1) then return false end
    local p, s = v.AbsolutePosition, v.AbsoluteSize
    local screenSize = Camera.ViewportSize
    if s.X <= 10 or s.Y <= 10 then return false end
    if p.X < 0 or p.Y < 0 or p.X + s.X > screenSize.X or p.Y + s.Y > screenSize.Y then return false end
    local parent = v.Parent
    while parent and parent:IsA("GuiObject") do
        if not parent.Visible then return false end
        parent = parent.Parent
    end
    return true
end

local function spamClickElement(v)
    while isTrulyVisible(v) and activeSpamTarget == v do
        local p, s = v.AbsolutePosition, v.AbsoluteSize
        local x, y = p.X + s.X/2, p.Y + s.Y/2 + 36
        VIM:SendMouseButtonEvent(x, y, 0, true, game, 1)
        VIM:SendMouseButtonEvent(x, y, 0, false, game, 1)
        task.wait(0.05) 
    end
    if activeSpamTarget == v then activeSpamTarget = nil end
end

local function evaluateHighestPriority()
    local highestTarget, highestIndex = nil, 99
    for v, _ in pairs(trackedButtons) do
        if isTrulyVisible(v) then
            local txt = v.Text:lower():gsub("%s+", "")
            for i, name in ipairs(list) do
                if txt == name and i < highestIndex then highestIndex = i highestTarget = v end
            end
        end
    end
    if highestTarget and activeSpamTarget ~= highestTarget then
        activeSpamTarget = highestTarget
        task.spawn(spamClickElement, highestTarget)
    end
end

local function bindTracking(v)
    if v:IsA("TextLabel") or v:IsA("TextButton") then
        local function updateCache()
            local txt = v.Text:lower():gsub("%s+", "")
            local isMatch = false
            for _, name in ipairs(list) do
                if txt == name then isMatch = true break end
            end
            if isMatch then trackedButtons[v] = true evaluateHighestPriority() else trackedButtons[v] = nil end
        end
        v:GetPropertyChangedSignal("Text"):Connect(updateCache)
        v:GetPropertyChangedSignal("Visible"):Connect(evaluateHighestPriority)
        v:GetPropertyChangedSignal("AbsolutePosition"):Connect(evaluateHighestPriority)
        updateCache()
    end
end

for _, desc in ipairs(PlayerGui:GetDescendants()) do bindTracking(desc) end

    end
    print("----------------------------------------------------------------------")
    print("Total Players: " .. #playerData)
    local lStat = SETTINGS.TARGET_LEVEL > 0 and tostring(SETTINGS.TARGET_LEVEL) or "OFF"
    local sStat = SETTINGS.TARGET_WINSTREAK > 0 and tostring(SETTINGS.TARGET_WINSTREAK) or "OFF"
    local rStat = SETTINGS.TARGET_RANK_ELO > 0 and tostring(SETTINGS.TARGET_RANK_ELO) or "OFF"
    local protStatus = SETTINGS.PROTECTION_ENABLED and "ON" or "OFF"
    print("Protection: " .. protStatus .. " (Level:" .. lStat .. " | Streak:" .. sStat .. " | Rank ELO:" .. rStat .. ")\n")
end

local function RefreshLeaderboard()
    if selfDestructed then return end
    for _, child in ipairs(ListFrame:GetChildren() or {}) do
        if child and child:IsA("TextLabel") then child:Destroy() end
    end
    local playerData = {}
    for _, plr in ipairs(Players:GetPlayers() or {}) do
        local data = GetPlayerStats(plr)
        if data then table.insert(playerData, data) end
    end
    table.sort(playerData, function(a, b)
        return (tonumber(a.Elo) or 0) > (tonumber(b.Elo) or 0)
    end)
    for i, data in ipairs(playerData) do
        if not data then break end
        local eloNum = tonumber(data.Elo) or 0
        local entry = Instance.new("TextLabel")
        entry.Size = UDim2.new(1, -4, 0, 22)
        entry.BackgroundTransparency = 1
        entry.Font = Enum.Font.Gotham
        entry.TextSize = 12
        entry.TextXAlignment = Enum.TextXAlignment.Left
        if data.Player == LocalPlayer then
            entry.TextColor3 = Color3.fromRGB(255, 215, 0)
            entry.Text = string.format("%d. %s  [YOU]", i, data.Name or "Unknown")
        else
            entry.TextColor3 = GetRankColor(eloNum)
            entry.Text = string.format("%d. %s", i, data.Name or "Unknown")
        end
        entry.Parent = ListFrame

        local statsEntry = Instance.new("TextLabel")
        statsEntry.Size = UDim2.new(0, 280, 0, 22)
        statsEntry.Position = UDim2.new(1, -284, 0, 0)
        statsEntry.BackgroundTransparency = 1
        statsEntry.Font = Enum.Font.Gotham
        statsEntry.TextSize = 11
        statsEntry.TextColor3 = Color3.fromRGB(160, 160, 190)
        statsEntry.TextXAlignment = Enum.TextXAlignment.Right
        statsEntry.Text = string.format("Levels: %s  |  %s (%s)  |  Winstreak: %s", data.Level or "?", GetRankName(eloNum), data.Elo or "?", data.Winstreak or "?")
        statsEntry.Parent = entry
    end
    ListFrame.CanvasSize = UDim2.new(0, 0, 0, #playerData * 24 + 10)
    if #playerData > 0 then PrintToConsole(playerData) end
    CheckForHighStats(playerData)
end

            duelsRemote:FireServer() 
        end

    elseif input.KeyCode == Enum.KeyCode.G then
        print("[BINDS] 'G' Pressed: Joining arcade...")
        triggerArcadeJoin()

        getgenv().WhitelistConfig.Enabled = not getgenv().WhitelistConfig.Enabled

        if getgenv().WhitelistConfig.Enabled then
            for _, player in ipairs(Players:GetPlayers()) do
                if player ~= LocalPlayer and not isWhitelisted(player.Name) then 
                    forceDisconnect() 
                    break 
                end
            end
        end

    elseif input.KeyCode == Enum.KeyCode.R then
        print("[BINDS] 'R' Pressed: Re-scanning and concealing map parts...")
        hideMapKeepPlayers()

    elseif input.KeyCode == Enum.KeyCode.T then
        SETTINGS.PROTECTION_ENABLED = not SETTINGS.PROTECTION_ENABLED
        UpdateStatusDot()
        print("[BINDS] 'T' Pressed: Target Stats Auto-Leave Protection is now -> " .. (SETTINGS.PROTECTION_ENABLED and "ENABLED" or "DISABLED"))

    elseif input.KeyCode == Enum.KeyCode.Y then
        print("[BINDS] 'Y' Pressed: Rejoining server instance...")
        rejoin()

    elseif input.KeyCode == Enum.KeyCode.F2 and not selfDestructed then
        selfDestructed = true
        print("[BINDS] 'F2' Pressed: Self-destruct initiated...")
        RivalsLeaderboard:Destroy()
        for _, v in ipairs(getgc and getgc() or {}) do
            if type(v) == "function" and getfenv and pcall(getfenv, v) then
                local env = getfenv(v)
                if env and env == fenv then 
                    pcall(function() setfenv(v, {}) end) 
                end
            end
        end
        print("[RIVALS] GUI destroyed. Script disabled.")

    elseif input.KeyCode == Enum.KeyCode.F4 then
        leaderboardVisible = not leaderboardVisible
        RivalsLeaderboard.Enabled = leaderboardVisible
        print("[BINDS] 'F4' Pressed: Leaderboard visibility toggled -> " .. tostring(leaderboardVisible))
    end
end)

local isRejoining = false

local function rejoinMatch()
    if isRejoining then return end
    isRejoining = true

    print("match ended (" .. START_TRACKING_AT .. "). rejoining game.")

    local success, errorMessage = pcall(function()
        TeleportService:TeleportToPlaceInstance(game.PlaceId, game.JobId, LocalPlayer)
    end)

    if not success then
        warn("rejoin failed: " .. tostring(errorMessage))
        isRejoining = false
    end
end

local function getObjectText(obj)
    local success, val = pcall(function() return obj.Value end)
    if success and val ~= nil then
        return tostring(val)
    elseif obj:IsA("TextLabel") or obj:IsA("TextBox") then
        return obj.Text
    end
    return nil
end

local function checkObject(obj)
    local text = getObjectText(obj)
    if text and string.gsub(text, "%s+", "") == START_TRACKING_AT then
        rejoinMatch()
    end
end

local function observeObject(obj)
    local changeSignal = obj:IsA("ValueBase") and obj.Changed or (obj:IsA("TextLabel") or obj:IsA("TextBox")) and obj:GetPropertyChangedSignal("Text")

    if changeSignal then
        changeSignal:Connect(function()
            checkObject(obj)
        end)
    end

    checkObject(obj)
end

print("rejoiner active (" .. START_TRACKING_AT .. ")")

PlayerGui.DescendantAdded:Connect(function(descendant)
    observeObject(descendant)
end)

task.spawn(function()
    local allDescendants = PlayerGui:GetDescendants()
    for i, descendant in ipairs(allDescendants) do
        observeObject(descendant)
        if i % 150 == 0 then task.wait() end
    end
end)
